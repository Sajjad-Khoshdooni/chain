\chapter{مروری بر مطالعات انجام شده}
\section{مقدمه}

در این فصل، به بررسی مطالعات و تحقیقات مرتبط با مقیاس‌پذیری بلاکچین می‌پردازیم. ابتدا بیتکوین و معماری آن را بررسی می‌کنیم، سپس راه‌حل‌های مختلف مقیاس‌پذیری را تحلیل می‌کنیم و در نهایت به مقایسه و ارزیابی این راه‌حل‌ها می‌پردازیم.

\section{بیتکوین و معماری آن}

بیتکوین که در سال 2008 توسط ساتوشی ناکاموتو معرفی شد، اولین پیاده‌سازی موفق از یک سیستم پول دیجیتال غیرمتمرکز است \cite{nakamoto2008bitcoin}. بیتکوین از یک معماری تک‌زنجیره‌ای استفاده می‌کند که در آن تمام تراکنش‌ها در یک زنجیره واحد ثبت می‌شوند.

\subsection{الگوریتم اجماع اثبات کار}

بیتکوین از الگوریتم اجماع اثبات کار \lr{(Proof of Work)} استفاده می‌کند. در این الگوریتم، ماینرها برای حل یک مسئله رمزنگاری رقابت می‌کنند و اولین ماینری که راه‌حل را پیدا کند، حق ایجاد بلوک جدید را دارد. این فرآیند نیاز به محاسبات قابل توجهی دارد و به طور متوسط هر 10 دقیقه یک بلوک جدید ایجاد می‌شود.

\subsection{محدودیت‌های مقیاس‌پذیری}

محدودیت اصلی بیتکوین در اندازه بلوک و زمان تولید بلوک نهفته است:
\begin{itemize}
    \item \textbf{اندازه بلوک}: حداکثر 1 مگابایت (که بعداً به 4 مگابایت در SegWit افزایش یافت)
    \item \textbf{زمان بلوک}: 10 دقیقه
    \item \textbf{نتیجه}: حداکثر 7 تراکنش در ثانیه
\end{itemize}

این محدودیت‌ها باعث می‌شود که بیتکوین در دوره‌های تقاضای بالا، با مشکل تأخیر و هزینه بالا مواجه شود.

\section{راه‌حل‌های لایه دوم}

\subsection{شبکه لایتنینگ}

شبکه لایتنینگ \lr{(Lightning Network)} که در سال 2016 توسط جوزف پون و تادریوس درایجا معرفی شد، یک راه‌حل لایه دوم برای مقیاس‌پذیری بیتکوین است \cite{poon2016lightning}. این شبکه کانال‌های پرداخت خارج از زنجیره را بین کاربران ایجاد می‌کند.

\subsubsection{مزایا}
\begin{itemize}
    \item پردازش هزاران تراکنش در ثانیه
    \item هزینه تراکنش بسیار پایین
    \item تأیید فوری تراکنش‌ها
\end{itemize}

\subsubsection{معایب}
\begin{itemize}
    \item نیاز به باز کردن و مدیریت کانال‌های پرداخت
    \item نیاز به نقدینگی در کانال‌ها
    \item ریسک تمرکز در صورت مدیریت نادرست
    \item پیچیدگی فنی برای کاربران عادی
\end{itemize}

\subsection{راه‌حل‌های مشابه}

راه‌حل‌های مشابه دیگری نیز وجود دارند مانند:
\begin{itemize}
    \item \textbf{Plasma}: برای اتریوم
    \item \textbf{Raiden Network}: برای اتریوم
    \item \textbf{State Channels}: کانال‌های حالت عمومی
\end{itemize}

\section{شاردینگ}

شاردینگ یک تکنیک پایگاه داده است که در بلاکچین برای تقسیم شبکه به بخش‌های کوچکتر (شارد) استفاده می‌شود. هر شارد می‌تواند به طور مستقل تراکنش‌ها را پردازش کند.

\subsection{شاردینگ در اتریوم 2.0}

اتریوم 2.0 از شاردینگ برای بهبود مقیاس‌پذیری استفاده می‌کند \cite{buterin2014ethereum}. در این معماری:
\begin{itemize}
    \item شبکه به 64 شارد تقسیم می‌شود
    \item هر شارد می‌تواند به طور مستقل تراکنش‌ها را پردازش کند
    \item ارتباط بین شاردها از طریق beacon chain انجام می‌شود
\end{itemize}

\subsection{چالش‌های شاردینگ}
\begin{itemize}
    \item \textbf{ارتباطات بین شارد}: نیاز به پروتکل‌های پیچیده برای ارتباط بین شاردها
    \item \textbf{امنیت}: هر شارد امنیت کمتری نسبت به کل شبکه دارد
    \item \textbf{پیچیدگی پیاده‌سازی}: نیاز به تغییرات اساسی در پروتکل
\end{itemize}

\section{مقدمه}
در این فصل به تشریح معماری سیستم بلاکچین پیشنهادی می‌پردازیم. معماری پیشنهادی، راه‌حلی نوآورانه برای حل مشکل مقیاس‌پذیری بلاکچین است که از ساختار زنجیره اصلی \lr{(Master Chain)} و زنجیره‌های جانبی \lr{(Side Chains)} استفاده می‌کند. این معماری با حفظ امنیت و غیرمتمرکز بودن، قادر به پردازش موازی تراکنش‌ها و افزایش قابل توجه توان عملیاتی شبکه است.

هدف اصلی این معماری، افزایش تعداد تراکنش‌های پردازش شده در واحد زمان \lr{(TPS)} نسبت به بلاکچین بیتکوین است، در حالی که ویژگی‌های اصلی بلاکچین مانند امنیت، شفافیت و غیرمتمرکز بودن حفظ شود.

\section{معماری کلی سیستم}

معماری پیشنهادی شامل دو لایه اصلی است:

\subsection{زنجیره اصلی \lr{(Master Chain)}}

زنجیره اصلی، ستون فقرات شبکه است که وظایف زیر را بر عهده دارد:

\begin{itemize}
    \item \textbf{الگوریتم اجماع}: از الگوریتم اثبات کار \lr{(Proof of Work)} مشابه بیتکوین استفاده می‌کند
    \item \textbf{زمان تولید بلوک}: هر 10 دقیقه یک بلوک جدید تولید می‌شود
    \item \textbf{محتوای بلوک}: ارجاعات به بلوک‌های زنجیره‌های جانبی را ذخیره می‌کند
    \item \textbf{امنیت}: با استفاده از الگوریتم \lr{SHA-256}، امنیت بالایی برای شبکه فراهم می‌کند
    \item \textbf{دشواری استخراج}: سطح دشواری قابل تنظیم است (معمولاً سطح بالا برای امنیت بیشتر)
\end{itemize}

ساختار یک بلوک در زنجیره اصلی به صورت زیر است:

\begin{latin}
\begin{equation}
MasterBlock = \{index, timestamp, references, previous\_hash, hash, nonce, difficulty\}
\end{equation}
\end{latin}

که در آن:
\begin{itemize}
    \item $index$: شماره بلوک
    \item $timestamp$: زمان تولید بلوک
    \item $references$: لیست ارجاعات به بلوک‌های زنجیره‌های جانبی
    \item $previous\_hash$: هش بلوک قبلی
    \item $hash$: هش بلوک فعلی
    \item $nonce$: عدد استفاده شده در استخراج
    \item $difficulty$: سطح دشواری استخراج
\end{itemize}

\subsection{زنجیره‌های جانبی \lr{(Side Chains)}}

زنجیره‌های جانبی، واحدهای پردازش موازی هستند که تراکنش‌ها را به صورت مستقل پردازش می‌کنند:

\begin{itemize}
    \item \textbf{تعداد}: تعداد زنجیره‌های جانبی قابل تنظیم و افزایش است (مقیاس‌پذیری افقی)
    \item \textbf{زمان تولید بلوک}: هر 1 دقیقه یک بلوک جدید تولید می‌شود
    \item \textbf{محتوای بلوک}: تراکنش‌های واقعی کاربران را پردازش می‌کند
    \item \textbf{دشواری استخراج}: سطح دشواری پایین‌تر از زنجیره اصلی (برای سرعت بیشتر)
    \item \textbf{اتصال به زنجیره اصلی}: هر بلوک زنجیره جانبی به یک بلوک در زنجیره اصلی مرتبط است
\end{itemize}

ساختار یک بلوک در زنجیره جانبی:

\begin{latin}
\begin{equation}
SideChainBlock = \{index, side\_chain\_id, timestamp, transactions, previous\_hash, parent\_master\_hash, hash, nonce\}
\end{equation}
\end{latin}

\section{الگوریتم‌های کلیدی}

\subsection{الگوریتم تخصیص تراکنش}

برای تخصیص تراکنش‌ها به زنجیره‌های جانبی، از یک الگوریتم قطعی و توزیع شده استفاده می‌شود که بر اساس UTXO \lr{(Unspent Transaction Output)} عمل می‌کند. پیاده‌سازی واقعی در Rust به صورت زیر است:

\begin{latin}
\begin{equation}
side\_chain\_id = numeric\_hash(UTXO) \bmod N
\end{equation}
\end{latin}

که در آن $numeric\_hash$ با استفاده از \lr{SHA-256} محاسبه می‌شود و 8 بایت اول به عدد صحیح 64 بیتی تبدیل می‌شود. در کد Rust:

\begin{latin}
\texttt{let utxo\_hash = utxo.numeric\_hash();}\\
\texttt{let side\_chain\_id = (utxo\_hash \% num\_side\_chains) as usize;}
\end{latin}

این الگوریتم دارای ویژگی‌های زیر است:

\begin{itemize}
    \item \textbf{قطعیت}: همیشه یک UTXO مشخص به همان زنجیره جانبی تخصیص داده می‌شود
    \item \textbf{توزیع یکنواخت}: استفاده از \lr{SHA-256} تضمین می‌کند که تراکنش‌ها به طور یکنواخت بین زنجیره‌های جانبی توزیع شوند
    \item \textbf{جلوگیری از دوبار خرج کردن}: با تخصیص قطعی، امکان دوبار خرج کردن یک UTXO وجود ندارد
    \item \textbf{پیچیدگی}: $O(1)$ - بسیار سریع و کارآمد (تنها یک عملیات هش و modulo)
\end{itemize}

\subsection{ماژول‌های پیشرفته در پیاده‌سازی}

پیاده‌سازی Rust شامل چهار ماژول پیشرفته برای افزایش قابلیت‌های سیستم است:

\subsubsection{ماژول رمزنگاری \lr{(Crypto Module)}}
\begin{itemize}
    \item \textbf{کیف پول \lr{(Wallet)}}: تولید کلیدهای خصوصی و عمومی با استفاده از \lr{SHA-256}
    \item \textbf{امضای دیجیتال \lr{(Signature)}}: امضا و تایید تراکنش‌ها با استفاده از ECDSA
    \item \textbf{مدیریت آدرس}: تولید آدرس‌های یکتا از کلیدهای عمومی
\end{itemize}

\subsubsection{ماژول اجماع \lr{(Consensus Module)}}
\begin{itemize}
    \item \textbf{حل فورک \lr{(Fork Resolution)}}: انتخاب زنجیره با بیشترین کار انجام شده
    \item \textbf{قطعیت \lr{(Finality)}}: تعیین قطعیت بلوک‌ها پس از 6 تایید (مشابه Bitcoin)
    \item \textbf{BFT}: پشتیبانی از تحمل خطای بیزانسی با فرمول $f < n/3$
\end{itemize}

\subsubsection{ماژول اعتبارسنجی (Validation Module)}
\begin{itemize}
    \item \textbf{مجموعه UTXO \lr{(UTXO Set)}}: ردیابی تمام UTXO های خرج نشده
    \item \textbf{جلوگیری از دوبار خرج کردن}: تشخیص و جلوگیری از double-spending
    \item \textbf{اعتبارسنجی تراکنش}: بررسی صحت مبلغ، امضا و کارمزد
\end{itemize}

\subsubsection{ماژول تراکنش بین زنجیره‌ای (Cross-Chain Module)}
\begin{itemize}
    \item \textbf{پل بین زنجیره‌ای (Cross-Chain Bridge)}: انتقال امن دارایی بین زنجیره‌های جانبی
    \item \textbf{قفل اتمیک \lr{(Atomic Swap)}}: تضمین همه یا هیچ در انتقالات
    \item \textbf{4 مرحله انتقال}: Initiated $\rightarrow$ Locked $\rightarrow$ Verified $\rightarrow$ Completed
\end{itemize}

این ماژول‌ها امکان ایجاد یک بلاکچین کامل و قابل استفاده در تولید را فراهم می‌کنند.

\section{فرآیند پردازش تراکنش}

فرآیند پردازش یک تراکنش در سیستم به صورت زیر است:

\begin{enumerate}
    \item \textbf{ایجاد تراکنش}: کاربر یک تراکنش جدید ایجاد می‌کند که شامل UTXO ورودی و خروجی است
    
    \item \textbf{تخصیص به زنجیره جانبی}: با استفاده از الگوریتم تخصیص، تراکنش به یک زنجیره جانبی مشخص ارسال می‌شود
    
    \item \textbf{اضافه شدن به Mempool}: تراکنش در mempool زنجیره جانبی مربوطه قرار می‌گیرد
    
    \item \textbf{استخراج بلوک زنجیره جانبی}: هر 1 دقیقه، یک بلوک جدید حاوی تراکنش‌های موجود در mempool استخراج می‌شود
    
    \item \textbf{ارجاع به زنجیره اصلی}: ارجاع به بلوک استخراج شده در زنجیره جانبی، در بلوک بعدی زنجیره اصلی ذخیره می‌شود
    
    \item \textbf{تایید نهایی}: پس از اضافه شدن ارجاع به زنجیره اصلی، تراکنش به طور نهایی تایید شده تلقی می‌شود
\end{enumerate}

\section{مزایای معماری پیشنهادی}

\subsection{مقیاص‌پذیری افقی}

یکی از مهم‌ترین مزایای این معماری، قابلیت مقیاس‌پذیری افقی است. با افزایش تعداد زنجیره‌های جانبی، ظرفیت کل شبکه به صورت خطی افزایش می‌یابد.

\textbf{نکته بسیار مهم:} با تنها یک زنجیره جانبی، عملکرد سیستم تقریباً مشابه بیتکوین است (فقط با بلوک‌های سریع‌تر). \textbf{مقیاس‌پذیری واقعی از پردازش موازی با چندین زنجیره جانبی حاصل می‌شود}.

\subsubsection{مدل تئوری}

برای محاسبه TPS تئوری از فرمول زیر استفاده می‌شود:

\begin{latin}
\begin{equation}
TPS_{theoretical} = \frac{M}{T_m} + \frac{N \times S}{T_s}
\end{equation}
\end{latin}

که در آن:
\begin{itemize}
    \item $M$: تعداد تراکنش در هر بلوک زنجیره اصلی
    \item $T_m$: زمان تولید بلوک زنجیره اصلی (600 ثانیه)
    \item $N$: تعداد زنجیره‌های جانبی
    \item $S$: تعداد تراکنش در هر بلوک زنجیره جانبی
    \item $T_s$: زمان تولید بلوک زنجیره جانبی (60 ثانیه)
\end{itemize}

این فرمول نشان می‌دهد که با افزایش $N$، TPS کل شبکه به صورت خطی افزایش می‌یابد.

\subsection{کاهش تاخیر}

با توجه به اینکه زنجیره‌های جانبی هر 1 دقیقه بلوک تولید می‌کنند (در مقایسه با 10 دقیقه در بیتکوین)، تاخیر تایید اولیه تراکنش‌ها به طور قابل توجهی کاهش می‌یابد.

\subsection{حفظ امنیت}

امنیت شبکه از طریق زنجیره اصلی که از الگوریتم اثبات کار با دشواری بالا استفاده می‌کند، حفظ می‌شود. تمام تغییرات در زنجیره‌های جانبی نهایتاً باید در زنجیره اصلی ثبت شوند.

\subsection{کارایی انرژی}

با توجه به اینکه زنجیره‌های جانبی با دشواری پایین‌تری کار می‌کنند، مصرف انرژی کلی شبکه در مقایسه با سیستمی که همه تراکنش‌ها را در یک زنجیره با دشواری بالا پردازش کند، بهینه‌تر است.

\section{محاسبه حداکثر TPS تئوری}

\subsection{روش‌شناسی محاسبه}

برای تعیین حداکثر ظرفیت تئوری سیستم، باید محدودیت‌های مختلف سیستم را شناسایی و تحلیل کنیم. پنج محدودیت اصلی عبارتند از:

\subsubsection{محدودیت اندازه بلوک}

با فرض اندازه بلوک 1 مگابایت (مشابه بیتکوین) و میانگین اندازه تراکنش 250 بایت:

\begin{latin}
\begin{equation}
Max\_Tx\_Per\_Block = \frac{Block\_Size}{Avg\_Tx\_Size} = \frac{1,048,576}{250} = 4,194
\end{equation}
\end{latin}

\subsubsection{محدودیت CPU و اعتبارسنجی}

با فرض زمان اعتبارسنجی 0.1 میلی‌ثانیه برای هر تراکنش:

\begin{latin}
\begin{equation}
Max\_Tx\_CPU = \frac{Block\_Time}{Validation\_Time} = \frac{60,000 ms}{0.1 ms} = 600,000
\end{equation}
\end{latin}

\subsubsection{محدودیت رمزنگاری}

با فرض زمان تایید امضای دیجیتال 0.5 میلی‌ثانیه:

\begin{latin}
\begin{equation}
Max\_Tx\_Crypto = \frac{60,000 ms}{0.5 ms} = 120,000
\end{equation}
\end{latin}

\subsubsection{تعیین گلوگاه سیستم}

گلوگاه (Bottleneck) سیستم، کوچکترین محدودیت است:

\begin{latin}
\begin{equation}
Bottleneck = \min(4,194, 600,000, 120,000, ...) = 4,194
\end{equation}
\end{latin}

بنابراین، \textbf{اندازه بلوک} محدودیت اصلی سیستم است.

\subsection{فرمول محاسبه TPS}

با استفاده از محدودیت شناسایی شده، TPS کل سیستم به صورت زیر محاسبه می‌شود:

\begin{latin}
\begin{equation}
TPS_{total} = \frac{M}{T_m} + \frac{N \times S}{T_s}
\end{equation}
\end{latin}

که در آن:
\begin{itemize}
    \item $M$: حداکثر تراکنش در هر بلوک زنجیره اصلی = 4,194
    \item $T_m$: زمان تولید بلوک زنجیره اصلی = 600 ثانیه
    \item $N$: تعداد زنجیره‌های جانبی
    \item $S$: حداکثر تراکنش در هر بلوک زنجیره جانبی = 4,194
    \item $T_s$: زمان تولید بلوک زنجیره جانبی = 60 ثانیه
\end{itemize}

با جایگذاری مقادیر:

\begin{latin}
\begin{align}
TPS_{master} &= \frac{4,194}{600} = 6.99 \\
TPS_{side} &= \frac{N \times 4,194}{60} = 69.90 \times N \\
TPS_{total} &= 6.99 + (69.90 \times N)
\end{align}
\end{latin}

\textbf{مشاهدات کلیدی:}

\begin{enumerate}
    \item \textbf{یک زنجیره جانبی ≈ بیتکوین}: با تنها 1 زنجیره جانبی، TPS تنها 11 برابر بیتکوین است (76.89 TPS). این نشان می‌دهد که بدون پردازش موازی، بهبود محدود است.
    
    \item \textbf{مقیاص‌پذیری خطی}: با افزودن هر زنجیره جانبی، دقیقاً 69.90 TPS به ظرفیت کل اضافه می‌شود. این رابطه کاملاً خطی است:
    \begin{latin}
    \begin{equation}
    \frac{\Delta TPS}{\Delta N} = 69.90 \text{ TPS per side chain}
    \end{equation}
    \end{latin}
    
    \item \textbf{مقیاس متوسط (10-100 زنجیره)}: برای کاربردهای تجاری، 10 تا 100 زنجیره جانبی می‌تواند 706 تا 6,997 TPS فراهم کند که برای اکثر برنامه‌ها کافی است.
    
    \item \textbf{مقیاس بزرگ (1000 زنجیره)}: با 1000 زنجیره جانبی، سیستم قادر به پردازش \textbf{69,907 TPS} است که تقریباً \textbf{10,000 برابر} بیتکوین است!
    
    \item \textbf{قابلیت گسترش نامحدود}: معماری پیشنهادی محدودیت نظری برای تعداد زنجیره‌های جانبی ندارد. تنها محدودیت‌ها، منابع سخت‌افزاری و شبکه هستند.
\end{enumerate}

\section{چرا یک زنجیره جانبی کافی نیست؟}

این بخش توضیح می‌دهد که چرا صرفاً داشتن یک زنجیره با بلوک‌های سریع‌تر نمی‌تواند مقیاص‌پذیری واقعی ایجاد کند.

\subsection{محدودیت‌های یک زنجیره}

با یک زنجیره جانبی، سیستم ما به صورت زیر عمل می‌کند:

\begin{latin}
\begin{equation}
TPS_{1\ chain} = \frac{M}{600} + \frac{1 \times S}{60} = \frac{4194}{600} + \frac{4194}{60} = 6.99 + 69.90 = 76.89
\end{equation}
\end{latin}

این فقط \textbf{11 برابر} بیتکوین است. دلایل:

\begin{enumerate}
    \item \textbf{محدودیت اندازه بلوک}: با بلوک 1MB و تراکنش 250 بایتی، حداکثر 4,194 تراکنش در هر بلوک جا می‌گیرد
    
    \item \textbf{محدودیت زمانی}: حتی با بلوک 1 دقیقه‌ای، TPS محدود به $4194/60 = 69.9$ است
    
    \item \textbf{مشکل دشواری}: اگر دشواری را خیلی کم کنیم (برای بلوک‌های سریع‌تر)، امنیت کاهش می‌یابد
    
    \item \textbf{محدودیت شبکه}: پهنای باند، تاخیر شبکه و propagation time مانع از بلوک‌های خیلی سریع می‌شود
\end{enumerate}

\textbf{نتیجه:} با یک زنجیره، ما فقط می‌توانیم زمان بلوک را کاهش دهیم (10 دقیقه → 1 دقیقه) که فقط 10x بهبود می‌دهد. این برای برنامه‌های مدرن کافی نیست!

\subsection{راه‌حل: پردازش موازی}

به جای افزایش سرعت یک زنجیره (که محدودیت فیزیکی دارد)، ما از \textbf{پردازش موازی} استفاده می‌کنیم:

\begin{itemize}
    \item \textbf{10 زنجیره موازی}: $706$ TPS (101x Bitcoin)
    \item \textbf{100 زنجیره موازی}: $6,997$ TPS (1,000x Bitcoin)
    \item \textbf{1000 زنجیره موازی}: $69,907$ TPS (10,000x Bitcoin)
\end{itemize}

هر زنجیره جانبی به صورت مستقل کار می‌کند و تراکنش‌های خود را پردازش می‌کند. زنجیره اصلی فقط ارجاعات را ذخیره می‌کند (overhead بسیار کم).

\subsection{هماهنگی بین زنجیره‌ای}

    \item استفاده از تراکنش‌های دو مرحله‌ای

    \item تایید متقابل بین زنجیره‌ها
\end{itemize}

\subsection{امنیت زنجیره‌های جانبی}

با توجه به دشواری پایین‌تر زنجیره‌های جانبی، امکان حمله 51\% بر روی یک زنجیره جانبی وجود دارد. راه‌حل‌های پیشنهادی:

\begin{itemize}
    \item الگوریتم تخصیص ماینر که هر ماینر را در چند زنجیره شرکت می‌دهد
    \item چک‌پوینت‌های دوره‌ای در زنجیره اصلی
    \item مکانیزم تشخیص و بازگشت تراکنش‌های مخرب
\end{itemize}
